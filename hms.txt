/* =========================================================
   1. DROP OLD TABLES
   ========================================================= */

DROP TABLE IF EXISTS Bill CASCADE;
DROP TABLE IF EXISTS Prescription CASCADE;
DROP TABLE IF EXISTS Treatment CASCADE;
DROP TABLE IF EXISTS Appointment CASCADE;
DROP TABLE IF EXISTS Patient_Contact CASCADE;
DROP TABLE IF EXISTS Ward CASCADE;
DROP TABLE IF EXISTS Doctor CASCADE;
DROP TABLE IF EXISTS Department CASCADE;
DROP TABLE IF EXISTS Patient CASCADE;


/* =========================================================
   2. CREATE PATIENT TABLE
   ========================================================= */

CREATE TABLE Patient (
    patient_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    dob DATE NOT NULL,
    blood_group VARCHAR(5) NOT NULL,
    gender VARCHAR(15) NOT NULL,
    address VARCHAR(255) NOT NULL
);


/* =========================================================
   3. CREATE DEPARTMENT TABLE
   ========================================================= */

CREATE TABLE Department (
    dept_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    dept_name VARCHAR(100) NOT NULL UNIQUE,
    hod_doctor_id INT,
    floor_no INT NOT NULL
);


/* =========================================================
   4. CREATE DOCTOR TABLE
   ========================================================= */

CREATE TABLE Doctor (
    doctor_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    specialization VARCHAR(100) NOT NULL,
    experience_years INT NOT NULL CHECK (experience_years >= 0),
    email VARCHAR(120) UNIQUE,
    dept_id INT,

    CONSTRAINT fk_doctor_department
        FOREIGN KEY (dept_id)
        REFERENCES Department(dept_id)
);


/* =========================================================
   5. ADD HOD FOREIGN KEY
   ========================================================= */

ALTER TABLE Department
ADD CONSTRAINT fk_department_hod
FOREIGN KEY (hod_doctor_id)
REFERENCES Doctor(doctor_id);


/* =========================================================
   6. CREATE PATIENT CONTACT TABLE
   ========================================================= */

CREATE TABLE Patient_Contact (
    contact_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    patient_id INT NOT NULL,
    contact_number VARCHAR(15) NOT NULL UNIQUE,
    contact_type VARCHAR(20) NOT NULL,

    CONSTRAINT fk_contact_patient
        FOREIGN KEY (patient_id)
        REFERENCES Patient(patient_id)
        ON DELETE CASCADE
);


/* =========================================================
   7. CREATE WARD TABLE
   ========================================================= */

CREATE TABLE Ward (
    ward_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ward_name VARCHAR(100) NOT NULL UNIQUE,
    capacity INT NOT NULL CHECK (capacity > 0),
    dept_id INT NOT NULL,

    CONSTRAINT fk_ward_department
        FOREIGN KEY (dept_id)
        REFERENCES Department(dept_id)
);


/* =========================================================
   8. CREATE APPOINTMENT TABLE
   ========================================================= */

CREATE TABLE Appointment (
    appointment_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    patient_id INT NOT NULL,
    doctor_id INT NOT NULL,
    appointment_date DATE NOT NULL,
    appointment_time TIME NOT NULL,
    status VARCHAR(20) NOT NULL,

    CONSTRAINT fk_appointment_patient
        FOREIGN KEY (patient_id)
        REFERENCES Patient(patient_id),

    CONSTRAINT fk_appointment_doctor
        FOREIGN KEY (doctor_id)
        REFERENCES Doctor(doctor_id)
);


/* =========================================================
   9. CREATE TREATMENT TABLE
   ========================================================= */

CREATE TABLE Treatment (
    treatment_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    appt_id INT NOT NULL UNIQUE,
    diagnosis VARCHAR(255) NOT NULL,
    treatment_date DATE NOT NULL,
    ward_id INT,

    CONSTRAINT fk_treatment_appointment
        FOREIGN KEY (appt_id)
        REFERENCES Appointment(appointment_id),

    CONSTRAINT fk_treatment_ward
        FOREIGN KEY (ward_id)
        REFERENCES Ward(ward_id)
);


/* =========================================================
   10. CREATE PRESCRIPTION TABLE
   ========================================================= */

CREATE TABLE Prescription (
    prescription_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    treatment_id INT NOT NULL,
    medicine_name VARCHAR(120) NOT NULL,
    dosage VARCHAR(60) NOT NULL,
    frequency VARCHAR(50) NOT NULL,
    duration INT NOT NULL CHECK (duration > 0),

    CONSTRAINT fk_prescription_treatment
        FOREIGN KEY (treatment_id)
        REFERENCES Treatment(treatment_id)
        ON DELETE CASCADE
);


/* =========================================================
   11. CREATE BILL TABLE
   ========================================================= */

CREATE TABLE Bill (
    bill_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    patient_id INT NOT NULL,
    treatment_id INT NOT NULL UNIQUE,
    consultation_fee DECIMAL(10,2) NOT NULL CHECK (consultation_fee >= 0),
    medicine_charges DECIMAL(10,2) NOT NULL CHECK (medicine_charges >= 0),
    bed_charges DECIMAL(10,2) NOT NULL CHECK (bed_charges >= 0),
    payment_status VARCHAR(20) NOT NULL,
    bill_date DATE NOT NULL,

    CONSTRAINT fk_bill_patient
        FOREIGN KEY (patient_id)
        REFERENCES Patient(patient_id),

    CONSTRAINT fk_bill_treatment
        FOREIGN KEY (treatment_id)
        REFERENCES Treatment(treatment_id)
);
